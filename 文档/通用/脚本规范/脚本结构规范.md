## 标准脚本模板

### 完整模板
```powershell
<#
.SYNOPSIS
    脚本简短描述

.DESCRIPTION
    脚本详细描述，说明功能和用途

.PARAMETER Force
    强制执行，跳过确认提示

.PARAMETER NoWait
    不等待用户按键确认（用于自动化调用）

.EXAMPLE
    .\script_name.ps1
    使用默认参数执行

.EXAMPLE
    .\script_name.ps1 -Force -NoWait
    强制执行，不等待用户确认

.NOTES
    作者: 林星晚
    日期: 2026-04-28
    模块: MonCore
#>

# ============================================================
# 脚本名称和用途描述
# 功能：详细说明脚本的功能
# 作者：林星晚
# 日期：2026-04-28
# ============================================================

# 参数定义
param(
    [switch]$Force = $false,
    [switch]$NoWait = $false,
    [string]$ConfigPath = ""
)

# 设置错误处理
$ErrorActionPreference = "Stop"

# 设置UTF-8编码
$OutputEncoding = [System.Text.Encoding]::UTF8
[Console]::OutputEncoding = [System.Text.Encoding]::UTF8

# 颜色定义
$ColorGreen = "Green"
$ColorYellow = "Yellow"
$ColorRed = "Red"
$ColorCyan = "Cyan"
$ColorMagenta = "Magenta"
$ColorGray = "Gray"
$ColorWhite = "White"

# 获取项目根目录
$ScriptDir = Split-Path -Parent $MyInvocation.MyCommand.Path
$ProjectRoot = Split-Path -Parent (Split-Path -Parent $ScriptDir)

# 显示标题
Write-Host "================================================" -ForegroundColor $ColorCyan
Write-Host "脚本标题" -ForegroundColor $ColorCyan
Write-Host "================================================" -ForegroundColor $ColorCyan
Write-Host ""

# 主要逻辑
try {
    # 1. 参数验证
    Write-Host "[1/N] 验证参数..." -ForegroundColor $ColorMagenta
    # 验证逻辑
    
    # 2. 环境检查
    Write-Host "[2/N] 检查环境..." -ForegroundColor $ColorMagenta
    # 检查逻辑
    
    # 3. 执行主要任务
    Write-Host "[3/N] 执行任务..." -ForegroundColor $ColorMagenta
    # 任务逻辑
    
    # 输出状态标识符
    Write-Host "[PROCESS_NAME:ProcessName]" -ForegroundColor $ColorCyan
    Write-Host "[SERVER_STATUS:SUCCESS]" -ForegroundColor $ColorGreen
    
    # 成功提示
    Write-Host ""
    Write-Host "================================================" -ForegroundColor $ColorGreen
    Write-Host "[✓] 任务完成！" -ForegroundColor $ColorGreen
    Write-Host "================================================" -ForegroundColor $ColorGreen
    Write-Host ""
    
} catch {
    # 输出错误状态标识符
    Write-Host "[SERVER_STATUS:FAILED]" -ForegroundColor $ColorRed
    
    # 错误处理
    Write-Host ""
    Write-Host "================================================" -ForegroundColor $ColorRed
    Write-Host "[✗] 任务失败！" -ForegroundColor $ColorRed
    Write-Host "================================================" -ForegroundColor $ColorRed
    Write-Host ""
    Write-Host "错误信息: $_" -ForegroundColor $ColorRed
    exit 1
}

# 等待用户确认（可选）
if (-not $NoWait) {
    Write-Host "按任意键退出..." -ForegroundColor $ColorGray
    $null = $Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown")
}
```

## 必需参数规范

### 通用参数
所有脚本都应该支持以下参数：

```powershell
param(
    [switch]$NoWait = $false,    # 不等待用户按键（用于自动化）
    [switch]$Force = $false,     # 强制执行（跳过确认）
    [switch]$Verbose = $false    # 详细输出
)
```

### 特定参数
根据脚本类型添加特定参数：

#### 启动脚本
```powershell
param(
    [switch]$NoWait = $false,
    [switch]$Force = $false,
    [switch]$NoClean = $false,      # 不清理端口和日志
    [switch]$UseRunserver = $false  # 使用 runserver 模式
)
```

#### 停止脚本
```powershell
param(
    [switch]$NoWait = $false,
    [switch]$Force = $false         # 强制停止
)
```

#### 状态脚本
```powershell
param(
    [switch]$NoWait = $false,
    [switch]$Detailed = $false      # 显示详细信息
)
```

## 配置文件集成

### 读取 .monconfig
```powershell
# 加载配置文件
$ConfigFile = Join-Path $ProjectRoot ".monconfig"
$ProcessName = "DefaultProcess"
$ServerPort = "8000"
$ServiceName = "DefaultService"

if (Test-Path $ConfigFile) {
    $ConfigContent = Get-Content $ConfigFile -Raw -Encoding UTF8
    
    # 读取进程名称
    if ($ConfigContent -match '\[process\][\s\S]*?NAME=(.+)') {
        $ProcessName = $Matches[1].Trim()
    }
    
    # 读取服务端口
    if ($ConfigContent -match '\[server\][\s\S]*?PORT=(.+)') {
        $ServerPort = $Matches[1].Trim()
    }
    
    # 读取服务名称
    if ($ConfigContent -match '\[service\][\s\S]*?NAME=(.+)') {
        $ServiceName = $Matches[1].Trim()
    }
    
    Write-Host "[✓] 配置加载成功" -ForegroundColor $ColorGreen
} else {
    Write-Host "[!] 配置文件不存在，使用默认值" -ForegroundColor $ColorYellow
}
```

## 错误处理规范

### 标准错误处理
```powershell
try {
    # 主要逻辑
    
} catch {
    # 输出错误状态标识符
    Write-Host "[SERVER_STATUS:FAILED]" -ForegroundColor $ColorRed
    
    # 错误信息显示

        Write-Host "按任意键退出..." -ForegroundColor $ColorGray
        $null = $Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown")
```
